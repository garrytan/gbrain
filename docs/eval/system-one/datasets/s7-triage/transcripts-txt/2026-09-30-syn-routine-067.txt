Python’s built-in `csv` module will handle quoted commas safely. Here is a script that reads `bank_export.csv` and writes `budget_import.csv`:

```python
import csv
from datetime import datetime
from decimal import Decimal

INPUT = "bank_export.csv"
OUTPUT = "budget_import.csv"

def parse_amount(value, row_number, field_name):
    value = value.strip()
    if not value:
        return None
    value = value.replace(",", "")
    try:
        amount = Decimal(value)
    except Exception as exc:
        raise ValueError(
            f"Row {row_number}: invalid {field_name} amount {value!r}"
        ) from exc
    if amount < 0:
        amount = abs(amount)
    return amount

with open(INPUT, "r", newline="", encoding="utf-8-sig") as source, \
     open(OUTPUT, "w", newline="", encoding="utf-8") as target:

    reader = csv.DictReader(source)
    required = {
        "Transaction Date", "Description", "Debit", "Credit",
        "Reference", "Category"
    }
    missing = required - set(reader.fieldnames or [])
    if missing:
        raise ValueError(f"Missing input columns: {sorted(missing)}")

    writer = csv.DictWriter(
        target,
        fieldnames=["Date", "Payee", "Amount", "Type", "Category", "Notes"]
    )
    writer.writeheader()

    for row_number, row in enumerate(reader, start=2):
        raw_date = row["Transaction Date"].strip()
        try:
            date = datetime.strptime(raw_date, "%m/%d/%Y").date()
        except ValueError as exc:
            raise ValueError(
                f"Row {row_number}: invalid date {raw_date!r}; "
                "expected MM/DD/YYYY"
            ) from exc

        debit = parse_amount(row["Debit"], row_number, "Debit")
        credit = parse_amount(row["Credit"], row_number, "Credit")

        if (debit is None) == (credit is None):
            raise ValueError(
                f"Row {row_number}: exactly one of Debit or Credit "
                "must contain an amount"
            )

        if debit is not None:
            amount = debit
            transaction_type = "Expense"
        else:
            amount = credit
            transaction_type = "Income"

        writer.writerow({
            "Date": date.isoformat(),
            "Payee": row["Description"].strip(),
            "Amount": f"{amount:.2f}",
            "Type": transaction_type,
            "Category": row["Category"].strip(),
            "Notes": row["Reference"].strip(),
        })

print(f"Wrote {OUTPUT}")
```

This deliberately stops on a malformed row instead of silently producing an import file with questionable data.

Yes. `datetime.strptime` accepts single-digit month and day values with `%m/%d/%Y` on common Python versions, but we can make date parsing explicit and remove currency symbols more carefully. Replace the parsing portions with this version:

```python
from datetime import datetime
from decimal import Decimal, InvalidOperation

def parse_date(value, row_number):
    value = value.strip()
    for date_format in ("%m/%d/%Y", "%Y-%m-%d"):
        try:
            return datetime.strptime(value, date_format).date()
        except ValueError:
            pass
    raise ValueError(
        f"Row {row_number}: invalid date {value!r}; "
        "expected M/D/YYYY or YYYY-MM-DD"
    )

def parse_amount(value, row_number, field_name):
    value = value.strip()
    if not value:
        return None

    cleaned = (
        value.replace("$", "")
             .replace(",", "")
             .strip()
    )

    # Support accounting-style negatives such as ($12.50).
    if cleaned.startswith("(") and cleaned.endswith(")"):
        cleaned = "-" + cleaned[1:-1].strip()

    try:
        amount = Decimal(cleaned)
    except InvalidOperation as exc:
        raise ValueError(
            f"Row {row_number}: invalid {field_name} amount {value!r}"
        ) from exc

    return abs(amount)
```

        "Transaction Date",
        "Description",
        "Reference",
        "Debit",
        "Credit",
        "Category",

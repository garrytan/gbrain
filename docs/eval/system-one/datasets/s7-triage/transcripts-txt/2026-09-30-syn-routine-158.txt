Here is a small Python script. It accepts one value per line, recognizes the unit suffix, prints the exact conversion to a sensible decimal precision, and adds an everyday rounded result. It supports the units in your list.

Save it as `convert_units.py`:

```python
import re

def convert(value, unit):
    unit = unit.lower()
    if unit in ("c", "°c"):
        return value * 9 / 5 + 32, "°F", round(value * 9 / 5 + 32)
    if unit in ("f", "°f"):
        result = (value - 32) * 5 / 9
        return result, "°C", round(result)
    if unit == "km":
        result = value * 0.621371
        return result, "mi", round(result, 1)
    if unit in ("mi", "mile", "miles"):
        result = value * 1.60934
        return result, "km", round(result, 1)
    if unit in ("m", "meter", "meters"):
        result = value * 3.28084
        return result, "ft", round(result)
    if unit in ("ft", "foot", "feet"):
        result = value * 0.3048
        return result, "m", round(result, 1)
    if unit in ("cm", "centimeter", "centimeters"):
        result = value / 2.54
        return result, "in", round(result, 1)
    if unit in ("in", "inch", "inches"):
        result = value * 2.54
        return result, "cm", round(result, 1)
    raise ValueError(f"Unsupported unit: {unit}")

print("Enter values such as 18 C, 72 F, or 3.5 km.")
print("Press Enter on a blank line to finish.")

while True:
    text = input("> ").strip()
    if not text:
        break

    match = re.fullmatch(r"([-+]?(?:\d+(?:\.\d*)?|\.\d+))\s*([A-Za-z°]+)", text)
    if not match:
        print("Format error. Use a number followed by a unit, such as -5 C.")
        continue

    try:
        value = float(match.group(1))
        unit = match.group(2)
        result, result_unit, rounded = convert(value, unit)
        print(f"Precise: {result:.6g} {result_unit}")
        print(f"Rounded: {rounded:g} {result_unit}")
    except ValueError as error:
        print(error)
```

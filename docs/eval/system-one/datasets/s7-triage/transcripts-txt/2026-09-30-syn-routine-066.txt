Here is a standard-library example. It treats the Pacific input as a local wall-clock time, prints the corresponding instant in UTC, Paris, and Kolkata, and checks local-time validity using round trips through UTC.

```python
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

PACIFIC = ZoneInfo("America/Los_Angeles")
PARIS = ZoneInfo("Europe/Paris")
KOLKATA = ZoneInfo("Asia/Kolkata")

def candidates(local_naive, zone):
    found = []
    for fold in (0, 1):
        aware = local_naive.replace(tzinfo=zone, fold=fold)
        utc = aware.astimezone(timezone.utc)
        round_trip = utc.astimezone(zone).replace(tzinfo=None)
        if round_trip == local_naive:
            item = (aware.utcoffset(), aware, utc)
            if not any(existing[0] == item[0] for existing in found):
                found.append(item)
    return sorted(found, key=lambda item: item[2])

def show_pacific(naive):
    options = candidates(naive, PACIFIC)
    print(f"\nPacific input: {naive:%Y-%m-%d %H:%M}")
    if not options:
        print("  NONEXISTENT local time")
        return
    if len(options) > 1:
        print("  AMBIGUOUS local time:")
    for _, pacific, utc in options:
        print("  Pacific:", pacific.strftime("%Y-%m-%d %H:%M:%S %Z %z"))
        print("  UTC:    ", utc.strftime("%Y-%m-%d %H:%M:%S %Z %z"))
        print("  Paris:  ", utc.astimezone(PARIS).strftime("%Y-%m-%d %H:%M:%S %Z %z"))
        print("  Kolkata:", utc.astimezone(KOLKATA).strftime("%Y-%m-%d %H:%M:%S %Z %z"))
```

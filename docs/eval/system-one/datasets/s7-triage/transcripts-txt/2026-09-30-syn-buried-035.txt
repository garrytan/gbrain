```python
import csv
import sys
from datetime import datetime
from decimal import Decimal

if len(sys.argv) != 3:
    raise SystemExit("Usage: python3 convert_budget_csv.py input.csv output.csv")

input_path, output_path = sys.argv[1], sys.argv[2]

with open(input_path, newline="", encoding="utf-8-sig") as source:
    reader = csv.DictReader(source)
    required = {"Date", "Merchant", "Debit", "Credit"}
    missing = required - set(reader.fieldnames or [])
    if missing:
        raise SystemExit(f"Missing columns: {', '.join(sorted(missing))}")

    with open(output_path, "w", newline="", encoding="utf-8") as target:
        writer = csv.DictWriter(
            target, fieldnames=["date", "amount", "description"]
        )
        writer.writeheader()

        for line_number, row in enumerate(reader, start=2):
            date_text = (row["Date"] or "").strip()
            merchant = (row["Merchant"] or "").strip()
            debit = (row["Debit"] or "").strip()
            credit = (row["Credit"] or "").strip()

            if not date_text or not merchant:
                raise SystemExit(f"Line {line_number}: missing date or merchant")

            try:
                date = datetime.strptime(date_text, "%m/%d/%Y").date()
                debit_value = Decimal(debit) if debit else Decimal("0")
                credit_value = Decimal(credit) if credit else Decimal("0")
            except Exception as error:
                raise SystemExit(f"Line {line_number}: {error}")

            if debit_value and credit_value:
                raise SystemExit(
                    f"Line {line_number}: both Debit and Credit contain values"
                )

            amount = credit_value - debit_value
            writer.writerow({
                "date": date.isoformat(),
                "amount": f"{amount:.2f}",
                "description": merchant,
            })
```

Use this revised script:

```python
import csv
import sys
from collections import Counter, defaultdict
from datetime import datetime
from decimal import Decimal, InvalidOperation

if len(sys.argv) != 2:
    raise SystemExit("Usage: python3 convert_budget_csv.py bank_export.csv")

input_path = sys.argv[1]
import_path = "budget_import.csv"
audit_path = "budget_audit.csv"
report_path = "budget_validation.txt"

required = {"Date", "Merchant", "Debit", "Credit", "Reference"}
rows = []
errors = []
monthly = defaultdict(lambda: {"count": 0, "total": Decimal("0")})

with open(input_path, newline="", encoding="utf-8-sig") as source:
    reader = csv.DictReader(source)
    missing = required - set(reader.fieldnames or [])
    if missing:
        raise SystemExit(f"Missing columns: {', '.join(sorted(missing))}")

    for line_number, row in enumerate(reader, start=2):
        date_text = (row["Date"] or "").strip()
        merchant = (row["Merchant"] or "").strip()
        debit_text = (row["Debit"] or "").strip()
        credit_text = (row["Credit"] or "").strip()
        reference = (row["Reference"] or "").strip()

        try:
            date = datetime.strptime(date_text, "%m/%d/%Y").date()
            debit = Decimal(debit_text) if debit_text else Decimal("0")
            credit = Decimal(credit_text) if credit_text else Decimal("0")

            if debit != 0 and credit != 0:
                raise ValueError("both Debit and Credit are populated")
            if not merchant:
                raise ValueError("Merchant is blank")

            if credit < 0:
                amount = abs(credit)
            elif debit != 0:
                amount = -abs(debit)
            else:
                amount = credit

            month = date.strftime("%Y-%m")
            monthly[month]["count"] += 1
            monthly[month]["total"] += amount

            rows.append({
                "date": date.isoformat(),
                "amount": f"{amount:.2f}",
                "description": merchant,
                "source_line": line_number,
                "reference": reference,
                "original_date": date_text,
                "original_debit": debit_text,
                "original_credit": credit_text,
                "category": (row["Category"] or "").strip(),
            })
        except (ValueError, InvalidOperation) as error:
            errors.append(f"Line {line_number}: {error}")

references = [row["reference"] for row in rows if row["reference"]]
duplicates = sorted(
    reference for reference, count in Counter(references).items() if count > 1
)

with open(report_path, "w", encoding="utf-8") as report:
    report.write(f"Input: {input_path}\n")
    report.write(f"Parsed rows: {len(rows)}\n")
    report.write(f"Rejected rows: {len(errors)}\n")
    report.write(f"Duplicate references: {len(duplicates)}\n\n")

    report.write("Monthly normalized totals:\n")
    for month in sorted(monthly):
        report.write(
            f"  {month}: {monthly[month]['count']} rows, "
            f"{monthly[month]['total']:.2f}\n"
        )

    if duplicates:
        report.write("\nDuplicate reference values:\n")
        for reference in duplicates:
            report.write(f"  {reference}\n")

    if errors:
        report.write("\nErrors:\n")
        for error in errors:
            report.write(f"  {error}\n")

if errors or duplicates:
    print(f"Validation found issues. See {report_path}. No output CSVs were written.")
    sys.exit(2)

with open(import_path, "w", newline="", encoding="utf-8") as target:
    writer = csv.DictWriter(
        target, fieldnames=["date", "amount", "description"]
    )
    writer.writeheader()
    writer.writerows(
        {key: row[key] for key in ("date", "amount", "description")}
        for row in rows
    )

with open(audit_path, "w", newline="", encoding="utf-8") as target:
    fields = [
        "source_line", "reference", "original_date", "original_debit",
        "original_credit", "category", "date", "amount", "description"
    ]
    writer = csv.DictWriter(target, fieldnames=fields)
    writer.writeheader()
    writer.writerows(rows)

print(f"Wrote {import_path}, {audit_path}, and {report_path}")
```

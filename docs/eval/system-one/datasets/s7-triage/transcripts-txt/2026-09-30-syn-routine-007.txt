I have about 300 rows, so I’d rather automate it. The file is called export.csv, and I’m on a computer with Python installed. Can you give me a script?

[assistant]
Yes. This script reads `export.csv`, converts dates from `MM/DD/YYYY` to `YYYY-MM-DD`, renames `Description` to `Payee`, preserves the other columns, and writes `budget_import.csv`.

```python
import csv
from datetime import datetime

input_file = "export.csv"
output_file = "budget_import.csv"

with open(input_file, "r", newline="", encoding="utf-8-sig") as source:
    reader = csv.DictReader(source)

    expected = {"Date", "Description", "Amount", "Category", "Account"}
    missing = expected - set(reader.fieldnames or [])
    if missing:
        raise ValueError(f"Missing columns: {', '.join(sorted(missing))}")

    fieldnames = ["Date", "Payee", "Amount", "Category", "Account"]

    with open(output_file, "w", newline="", encoding="utf-8") as target:
        writer = csv.DictWriter(target, fieldnames=fieldnames)
        writer.writeheader()

        for line_number, row in enumerate(reader, start=2):
            try:
                date_value = datetime.strptime(
                    row["Date"].strip(), "%m/%d/%Y"
                ).strftime("%Y-%m-%d")
            except ValueError as error:
                raise ValueError(
                    f"Invalid date on CSV line {line_number}: "
                    f"{row['Date']!r}"
                ) from error

            writer.writerow({
                "Date": date_value,
                "Payee": row["Description"].strip(),
                "Amount": row["Amount"].strip(),
                "Category": row["Category"].strip(),
                "Account": row["Account"].strip(),
            })

print(f"Wrote {output_file}")
```

[assistant]
Yes. The amount cleaner can return an empty string for blank values, while the main loop can collect rows whose category is blank. The script below writes the converted file and prints a review list after processing. It does not invent categories.

```python
import csv
from datetime import datetime
from decimal import Decimal, InvalidOperation

input_file = "export.csv"
output_file = "budget_import.csv"

def clean_amount(value, line_number):
    cleaned = value.strip().replace("$", "").replace(",", "")
    if cleaned == "":
        return ""

    try:
        Decimal(cleaned)
    except InvalidOperation as error:
        raise ValueError(
            f"Invalid amount on CSV line {line_number}: {value!r}"
        ) from error

    return cleaned

missing_categories = []

with open(input_file, "r", newline="", encoding="utf-8-sig") as source:
    reader = csv.DictReader(source)

    expected = {"Date", "Description", "Amount", "Category", "Account"}
    missing = expected - set(reader.fieldnames or [])
    if missing:
        raise ValueError(f"Missing columns: {', '.join(sorted(missing))}")

    fieldnames = ["Date", "Payee", "Amount", "Category", "Account"]

    with open(output_file, "w", newline="", encoding="utf-8") as target:
        writer = csv.DictWriter(target, fieldnames=fieldnames)
        writer.writeheader()

        for line_number, row in enumerate(reader, start=2):
            try:
                date_value = datetime.strptime(
                    row["Date"].strip(), "%m/%d/%Y"
                ).strftime("%Y-%m-%d")
            except ValueError as error:
                raise ValueError(
                    f"Invalid date on CSV line {line_number}: "
                    f"{row['Date']!r}"
                ) from error

            category = row["Category"].strip()
            if not category:
                missing_categories.append({
                    "line": line_number,
                    "payee": row["Description"].strip(),
                    "date": date_value,
                })

            writer.writerow({
                "Date": date_value,
                "Payee": row["Description"].strip(),
                "Amount": clean_amount(row["Amount"], line_number),
                "Category": category,
                "Account": row["Account"].strip(),
            })

print(f"Wrote {output_file}")

if missing_categories:
    print("\nRows with missing categories:")
    for item in missing_categories:
        print(
            f"CSV line {item['line']}: "
            f"{item['date']} | {item['payee']}"
        )
else:
    print("No rows with missing categories.")
```

A blank amount will remain blank in the output. Rows with blank categories will still be written, but the script will list them so you can fill them before importing.

Yes. The script can create `missing_categories.csv` with the source line number, date, payee, amount, and a blank category column. You can fill in that file, but it will not automatically merge those edits back into the import file. For a simple workflow, the script can instead write all converted rows to a review file, which you edit once before importing.

Here is a version that creates both files:

```python
import csv
from datetime import datetime
from decimal import Decimal, InvalidOperation

input_file = "export.csv"
output_file = "budget_import.csv"
report_file = "missing_categories.csv"

def clean_amount(value, line_number):
    cleaned = value.strip().replace("$", "").replace(",", "")
    if cleaned == "":
        return ""

    try:
        Decimal(cleaned)
    except InvalidOperation as error:
        raise ValueError(
            f"Invalid amount on CSV line {line_number}: {value!r}"
        ) from error

    return cleaned

missing_categories = []

with open(input_file, "r", newline="", encoding="utf-8-sig") as source:
    reader = csv.DictReader(source)
    expected = {"Date", "Description", "Amount", "Category", "Account"}
    missing = expected - set(reader.fieldnames or [])
    if missing:
        raise ValueError(f"Missing columns: {', '.join(sorted(missing))}")

    fieldnames = ["Date", "Payee", "Amount", "Category", "Account"]

    with open(output_file, "w", newline="", encoding="utf-8") as target:
        writer = csv.DictWriter(target, fieldnames=fieldnames)
        writer.writeheader()

        for line_number, row in enumerate(reader, start=2):
            try:
                date_value = datetime.strptime(
                    row["Date"].strip(), "%m/%d/%Y"
                ).strftime("%Y-%m-%d")
            except ValueError as error:
                raise ValueError(
                    f"Invalid date on CSV line {line_number}: "
                    f"{row['Date']!r}"
                ) from error

            amount = clean_amount(row["Amount"], line_number)
            category = row["Category"].strip()

            writer.writerow({
                "Date": date_value,
                "Payee": row["Description"].strip(),
                "Amount": amount,
                "Category": category,
                "Account": row["Account"].strip(),
            })

            if not category:
                missing_categories.append({
                    "SourceLine": line_number,
                    "Date": date_value,
                    "Payee": row["Description"].strip(),
                    "Amount": amount,
                    "Category": "",
                    "Account": row["Account"].strip(),
                })

with open(report_file, "w", newline="", encoding="utf-8") as report:
    report_fields = [
        "SourceLine", "Date", "Payee", "Amount", "Category", "Account"
    ]
    writer = csv.DictWriter(report, fieldnames=report_fields)
    writer.writeheader()
    writer.writerows(missing_categories)

print(f"Wrote {output_file}")
print(f"Wrote {report_file} with {len(missing_categories)} rows")
```

The separate report is useful for reference, but editing it alone will not update `budget_import.csv`. If you want category edits to flow back automatically, use a single review CSV as the editable source and generate the final import file from that after you finish editing.

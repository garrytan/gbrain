```python
from pathlib import Path
from datetime import datetime
from PIL import Image
from PIL.ExifTags import TAGS

PHOTO_EXTENSIONS = {".jpg", ".jpeg", ".png", ".tif", ".tiff", ".webp"}

def get_taken_time(path):
    try:
        with Image.open(path) as image:
            exif = image.getexif()
            for tag_id, value in exif.items():
                tag_name = TAGS.get(tag_id, tag_id)
                if tag_name == "DateTimeOriginal":
                    return datetime.strptime(value, "%Y:%m:%d %H:%M:%S")
    except Exception as error:
        print(f"Could not read {path.name}: {error}")
    return None

def main():
    folder = Path("photos")
    used_names = set()

    for path in sorted(folder.iterdir()):
        if not path.is_file() or path.suffix.lower() not in PHOTO_EXTENSIONS:
            continue

        taken = get_taken_time(path)
        if taken is None:
            print(f"Skipping {path.name}: no DateTimeOriginal")
            continue

        base = taken.strftime("%Y-%m-%d_%H-%M-%S")
        extension = path.suffix.lower()
        new_name = f"{base}{extension}"
        counter = 1

        while new_name in used_names or (folder / new_name).exists():
            new_name = f"{base}_{counter:02d}{extension}"
            counter += 1

        used_names.add(new_name)
        destination = folder / new_name
        print(f"{path.name} -> {new_name}")
        path.rename(destination)

if __name__ == "__main__":
    main()
```

What should I do with the photos that do not have `DateTimeOriginal`?

[assistant]
Those files may still have another EXIF date, or they may have no embedded date at all. You can use this fallback order:

1. `DateTimeOriginal`, usually the capture time.
2. `DateTimeDigitized`, often close to the capture time.
3. `DateTime`, a general image-modified or saved time.
4. The file’s filesystem modification time, if you explicitly want a fallback.

The first three are stored in EXIF. The filesystem time can be changed by copying files, downloading them, or moving them between systems, so it is less reliable.

Here is a replacement for `get_taken_time` that checks the three EXIF fields:

```python
def get_taken_time(path):
    preferred_tags = [
        "DateTimeOriginal",
        "DateTimeDigitized",
        "DateTime",
    ]

    try:
        with Image.open(path) as image:
            exif = image.getexif()
            values = {}

            for tag_id, value in exif.items():
                tag_name = TAGS.get(tag_id, tag_id)
                if tag_name in preferred_tags:
                    if isinstance(value, bytes):
                        value = value.decode(errors="replace")
                    values[tag_name] = value

            for tag_name in preferred_tags:
                value = values.get(tag_name)
                if not value:
                    continue
                try:
                    return datetime.strptime(value, "%Y:%m:%d %H:%M:%S")
                except ValueError:
                    print(
                        f"Ignoring invalid {tag_name} in "
                        f"{path.name}: {value!r}"
                    )

    except Exception as error:
        print(f"Could not read {path.name}: {error}")

    return None
```

Yes, the existing-name behavior is expected, but there is a subtle issue in the current collision check.

The line:

```python
while new_name in used_names or (FOLDER / new_name).exists():
```

protects an incoming file from overwriting an existing path. However, if an existing file is itself one of the files being considered for renaming, the script still treats its current name as occupied. That can create unnecessary suffixes during a rerun or when several files are being renamed.

Also, after a successful run, the script will scan the renamed files again. Files with ordinary date-based names still have photo extensions, so they are eligible. Files with `NOEXIF_` names are eligible too. With the current logic, they may be renamed again, usually to the same base name or a suffixed name depending on filesystem timestamps and collisions.

A safer version should:

- Ignore files already matching the generated naming pattern.
- Reserve names of files that are not being renamed.
- Use a temporary two-step rename when a source and destination overlap.
- Keep dry-run output separate from actual changes.

Here is a complete version. It skips names generated by the script, plans all changes before renaming, checks collisions against files that will remain in place, and uses temporary names during the real rename so a destination cannot block a source.

```python
from pathlib import Path
from datetime import datetime
import re
from PIL import Image
from PIL.ExifTags import TAGS

PHOTO_EXTENSIONS = {".jpg", ".jpeg", ".png", ".tif", ".tiff", ".webp"}
FOLDER = Path("/Users/me/Desktop/Vacation Photos")
DRY_RUN = True

GENERATED_NAME = re.compile(
    r"^(?:NOEXIF_)?\d{4}-\d{2}-\d{2}_\d{2}-\d{2}-\d{2}"
    r"(?:_\d{2})?\.[^.]+$",
    re.IGNORECASE,
)


def get_date_and_source(path):
    preferred_tags = (
        "DateTimeOriginal",
        "DateTimeDigitized",
        "DateTime",
    )

    try:
        with Image.open(path) as image:
            exif = image.getexif()

            for wanted_tag in preferred_tags:
                for tag_id, value in exif.items():
                    tag_name = TAGS.get(tag_id, tag_id)
                    if tag_name != wanted_tag:
                        continue

                    if isinstance(value, bytes):
                        value = value.decode(errors="replace")

                    try:
                        parsed = datetime.strptime(
                            value, "%Y:%m:%d %H:%M:%S"
                        )
                        return parsed, "EXIF"
                    except (TypeError, ValueError):
                        print(
                            f"Ignoring invalid {tag_name} in "
                            f"{path.name}: {value!r}"
                        )

    except Exception as error:
        print(f"Could not read EXIF from {path.name}: {error}")

    try:
        modified = datetime.fromtimestamp(path.stat().st_mtime)
        return modified, "NOEXIF"
    except OSError as error:
        print(f"Could not read file time for {path.name}: {error}")
        return None, None


def make_unique_name(folder, base, extension, reserved):
    candidate = f"{base}{extension}"
    counter = 1

    while candidate in reserved or (folder / candidate).exists():
        candidate = f"{base}_{counter:02d}{extension}"
        counter += 1

    reserved.add(candidate)
    return candidate


def main():
    if not FOLDER.is_dir():
        raise SystemExit(f"Folder does not exist: {FOLDER}")

    candidates = []

    for path in sorted(FOLDER.iterdir()):
        if not path.is_file():
            continue
        if path.suffix.lower() not in PHOTO_EXTENSIONS:
            continue
        if GENERATED_NAME.fullmatch(path.name):
            print(f"Skipping already generated name: {path.name}")
            continue

        candidates.append(path)

    candidate_paths = set(candidates)
    reserved = {
        path.name
        for path in FOLDER.iterdir()
        if path.is_file() and path not in candidate_paths
    }

    changes = []

    for path in candidates:
        taken, source = get_date_and_source(path)

        if taken is None:
            print(f"Skipping {path.name}: no usable date")
            continue

        timestamp = taken.strftime("%Y-%m-%d_%H-%M-%S")
        prefix = "" if source == "EXIF" else "NOEXIF_"
        base = prefix + timestamp
        extension = path.suffix.lower()

        new_name = make_unique_name(
            FOLDER,
            base,
            extension,
            reserved,
        )

        if new_name == path.name:
            print(f"Skipping unchanged name: {path.name}")
            continue

        changes.append((path, FOLDER / new_name, source))

    print("\nPlanned changes:")
    for source_path, destination, source in changes:
        print(
            f"{source_path.name} -> {destination.name} "
            f"[{source}]"
        )

    print(f"\nPhotos to rename: {len(changes)}")
    print(f"Dry run: {DRY_RUN}")

    if DRY_RUN or not changes:
        return

    temporary_changes = []

    for index, (source_path, destination, source) in enumerate(changes):
        temporary_name = (
            f".rename_photos_tmp_{index:06d}{source_path.suffix.lower()}"
        )
        temporary_path = FOLDER / temporary_name

        while temporary_path.exists():
            index += 1
            temporary_name = (
                f".rename_photos_tmp_{index:06d}"
                f"{source_path.suffix.lower()}"
            )
            temporary_path = FOLDER / temporary_name

        source_path.rename(temporary_path)
        temporary_changes.append((temporary_path, destination))

    for temporary_path, destination in temporary_changes:
        temporary_path.rename(destination)

    print("Renaming complete.")


if __name__ == "__main__":
    main()
```

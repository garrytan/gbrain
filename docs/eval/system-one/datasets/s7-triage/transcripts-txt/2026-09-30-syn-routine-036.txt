One concern: Python’s `replace(tzinfo=...)` can be questionable around a DST transition. Could you revise the script to detect ambiguous or nonexistent Los Angeles local times, even though this particular range is not near the transition?

[assistant]
Yes. A practical standard-library approach is to round-trip both `fold=0` and `fold=1` through UTC. For a valid unambiguous time, both folds round-trip to the same wall time and normally have the same offset. For an ambiguous time, both round-trip but produce different offsets. For a nonexistent time, neither fold round-trips to the original wall time.

```python
from datetime import datetime, timedelta, timezone
from zoneinfo import ZoneInfo

LA = ZoneInfo("America/Los_Angeles")
UTC = timezone.utc
LONDON = ZoneInfo("Europe/London")
KOLKATA = ZoneInfo("Asia/Kolkata")

def classify_local(naive, zone):
    candidates = []

    for fold in (0, 1):
        aware = naive.replace(tzinfo=zone, fold=fold)
        instant = aware.astimezone(UTC)
        round_trip = instant.astimezone(zone)

        if round_trip.replace(tzinfo=None) == naive:
            candidates.append(aware)

    # Remove duplicate candidates, which occur for ordinary times.
    unique = {}
    for item in candidates:
        key = item.astimezone(UTC)
        unique[key] = item
    candidates = list(unique.values())

    if not candidates:
        return "nonexistent", []
    if len(candidates) > 1:
        return "ambiguous", sorted(candidates, key=lambda x: x.astimezone(UTC))
    return "valid", candidates

def in_window(dt, start_minute, end_minute):
    minute = dt.hour * 60 + dt.minute
    return start_minute <= minute <= end_minute

start = datetime(2026, 3, 30, 6, 0)
end = datetime(2026, 3, 30, 10, 0)

current = start
while current <= end:
    status, candidates = classify_local(current, LA)

    if status != "valid":
        print(f"{current:%Y-%m-%d %H:%M} Los Angeles: {status}")
        current += timedelta(minutes=30)
        continue

    la = candidates[0]
    utc = la.astimezone(UTC)
    london = utc.astimezone(LONDON)
    kolkata = utc.astimezone(KOLKATA)

    fits = (
        in_window(la, 6 * 60, 18 * 60)
        and in_window(london, 8 * 60, 18 * 60)
        and in_window(kolkata, 8 * 60, 19 * 60)
    )

    print(
        f"{la:%a %Y-%m-%d %H:%M %Z} | "
        f"{utc:%a %Y-%m-%d %H:%M UTC} | "
        f"{london:%a %Y-%m-%d %H:%M %Z} | "
        f"{kolkata:%a %Y-%m-%d %H:%M %Z} | "
        f"{'YES' if fits else 'NO'}"
    )

    current += timedelta(minutes=30)
```

For a fall-back time, `classify_local` returns two candidates, one for each occurrence. For a spring-forward gap, it returns `nonexistent`, so the script does not silently convert a wall time that never occurs.

        instant = aware.astimezone(UTC)
        if instant.astimezone(zone).replace(tzinfo=None) == naive:
            candidates.append(aware)

    unique = {}
    for candidate in candidates:
        unique[candidate.astimezone(UTC)] = candidate

        return "nonexistent", []

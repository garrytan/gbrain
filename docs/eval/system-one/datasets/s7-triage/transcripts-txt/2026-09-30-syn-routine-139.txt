from datetime import datetime

source_file = "bank_export.csv"

        )

    with open(output_file, "w", newline="", encoding="utf-8") as target:
        writer = csv.DictWriter(target, fieldnames=output_fields)
        writer.writeheader()

        for line_number, row in enumerate(reader, start=2):
            raw_date = row["Transaction Date"].strip()
            try:
                normalized_date = datetime.strptime(
                    raw_date, "%m/%d/%Y"
                ).strftime("%Y-%m-%d")
            except ValueError as error:
                raise ValueError(
                    f"Invalid date on input line {line_number}: {raw_date!r}"

                "Date": normalized_date,
                "Description": row["Description"],
                "Amount": row["Amount"].strip(),
                "Category": row["Category"].strip(),
                "Account": row["Account"].strip(),

[assistant]
It will preserve a negative value such as `-42.17`, but it will not safely handle an unquoted amount containing a comma. A properly quoted CSV value such as `"$1,234.56"` will be read as one field, but the dollar sign and comma may still be rejected by the app.

You can normalize amounts to a plain decimal number. Add this function near the top of the script:

```python
from decimal import Decimal, InvalidOperation

def normalize_amount(raw_amount, line_number):
    cleaned = raw_amount.strip().replace("$", "").replace(",", "")

    if cleaned == "":
        raise ValueError(
            f"Missing amount on input line {line_number}"
        )

    try:
        value = Decimal(cleaned)
    except InvalidOperation as error:
        raise ValueError(
            f"Invalid amount on input line {line_number}: {raw_amount!r}"
        ) from error

    return format(value, "f")
```

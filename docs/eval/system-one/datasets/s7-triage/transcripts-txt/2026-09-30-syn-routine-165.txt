```python
from pathlib import Path
from datetime import datetime
from PIL import Image
import argparse
import re
import sys

IMAGE_EXTENSIONS = {".jpg", ".jpeg", ".JPG", ".JPEG"}

def capture_datetime(path):
    try:
        with Image.open(path) as image:
            exif = image.getexif()
            value = exif.get(36867) or exif.get(306)  # DateTimeOriginal or DateTime
            if not value:
                return None
            return datetime.strptime(str(value), "%Y:%m:%d %H:%M:%S")
    except Exception as error:
        print(f"Could not read EXIF from {path.name}: {error}", file=sys.stderr)
        return None

def available_name(folder, desired, reserved):
    candidate = desired
    counter = 1
    while candidate.exists() or candidate in reserved:
        candidate = desired.with_name(
            f"{desired.stem}_{counter:03d}{desired.suffix}"
        )
        counter += 1
    reserved.add(candidate)
    return candidate

parser = argparse.ArgumentParser()
parser.add_argument("folder", type=Path)
parser.add_argument("--apply", action="store_true")
args = parser.parse_args()

folder = args.folder
if not folder.is_dir():
    raise SystemExit(f"Not a folder: {folder}")

files = [
    path for path in folder.iterdir()
    if path.is_file() and path.suffix in IMAGE_EXTENSIONS
]

dated = []
skipped = []

for path in files:
    taken = capture_datetime(path)
    if taken is None:
        skipped.append(path)
    else:
        dated.append((taken, path))

dated.sort(key=lambda item: (item[0], item[1].name.lower()))

reserved = set()
changes = []

for number, (taken, old_path) in enumerate(dated, start=1):
    new_name = f"{taken:%Y-%m-%d}_{number:03d}.jpg"
    new_path = available_name(folder, new_name, reserved)
    changes.append((old_path, new_path))

print(f"Found {len(files)} image(s).")
print(f"With capture dates: {len(dated)}")
print(f"Skipped without usable EXIF date: {len(skipped)}")

for old_path, new_path in changes:
    print(f"{old_path.name} -> {new_path.name}")

if skipped:
    print("\nSkipped files:")
    for path in skipped:
        print(f"- {path.name}")

if not args.apply:
    print("\nPreview only. Re-run with --apply to perform these renames.")
else:
    for old_path, new_path in changes:
        old_path.rename(new_path)
    print("\nRenaming complete.")
```

const fs = require('fs');

function canonical(value) {

}

[assistant]
Yes. Compare the `packages` maps by path and print additions, removals, and selected field changes. This avoids dumping the entire lockfile:

```bash
node <<'NODE'
const fs = require('fs');

const files = {
  ours: '/tmp/package-lock.ours.json',
  theirs: '/tmp/package-lock.theirs.json',
  generated: 'package-lock.json'
};

const locks = Object.fromEntries(
  Object.entries(files).map(([name, file]) => [
    name,
    JSON.parse(fs.readFileSync(file))
  ])
);

function packages(lock) {
  return lock.packages || {};
}

function value(obj, key) {
  return obj && Object.prototype.hasOwnProperty.call(obj, key)
    ? JSON.stringify(obj[key])
    : '<absent>';
}

for (const side of ['ours', 'theirs']) {
  const before = packages(locks[side]);
  const after = packages(locks.generated);
  const paths = new Set([...Object.keys(before), ...Object.keys(after)]);

  const added = [];
  const removed = [];
  const changed = [];

  for (const path of [...paths].sort()) {
    if (!(path in before)) {
      added.push(path);
      continue;
    }
    if (!(path in after)) {
      removed.push(path);
      continue;
    }

    const fields = ['version', 'resolved', 'integrity', 'link', 'dev', 'optional'];
    const differences = fields.filter(field =>
      value(before[path], field) !== value(after[path], field)
    );

    if (differences.length) changed.push({path, fields: differences});
  }

  console.log(`\n=== generated versus ${side} ===`);
  console.log(`added: ${added.length}`);
  console.log(added.slice(0, 100).join('\n'));
  console.log(`removed: ${removed.length}`);
  console.log(removed.slice(0, 100).join('\n'));
  console.log(`changed: ${changed.length}`);
  for (const item of changed.slice(0, 100)) {
    console.log(`${item.path}: ${item.fields.join(', ')}`);
  }
}
NODE
```

The output is capped at 100 paths per category to keep it readable. To inspect one particular path in full, use:

```bash
node -e '
const l=require("./package-lock.json");
console.log(JSON.stringify(l.packages["node_modules/example-package"], null, 2));
'
```

Replace the path with one from the report. For a root dependency, inspect `packages[""]`.

Also check whether the generated lockfile uses a different registry URL or npm metadata convention:

```bash
grep -nE '"resolved"|"integrity"|"registry"' package-lock.json | head -40
```

Do not treat every integrity change as an error; regenerated lockfiles can update metadata while preserving the same package versions.

Normalize the `Amount` column to plain decimal numbers: `$1,250.00` becomes `1250.00`, and `($4.75)` becomes `-4.75`. Because commas can also occur inside quoted descriptions, use the CSV parser again. Here is a complete script that handles the date rename and amount cleanup together:

```bash
cd ~/Downloads
python3 - <<'PY'
import csv
from datetime import datetime
from decimal import Decimal, InvalidOperation

source = "transactions_export.csv"
target = "transactions_export_ready.csv"

def clean_amount(value):
    text = value.strip()
    if text.startswith("(") and text.endswith(")"):
        text = "-" + text[1:-1]
    text = text.replace("$", "").replace(",", "").strip()
    try:
        return format(Decimal(text), "f")
    except InvalidOperation:
        raise ValueError(f"Invalid amount: {value!r}")

with open(source, newline="", encoding="utf-8-sig") as infile, \
     open(target, "w", newline="", encoding="utf-8") as outfile:
    reader = csv.DictReader(infile)
    if not reader.fieldnames:
        raise SystemExit("The file has no header row")
    if "Date" not in reader.fieldnames:
        raise SystemExit("Could not find Date column")
    if "Amount" not in reader.fieldnames:
        raise SystemExit("Could not find Amount column")

    fieldnames = [
        "Transaction Date" if name == "Date" else name
        for name in reader.fieldnames
    ]
    writer = csv.DictWriter(outfile, fieldnames=fieldnames)
    writer.writeheader()

    for line_number, row in enumerate(reader, start=2):
        try:
            row["Date"] = datetime.strptime(
                row["Date"].strip(), "%m/%d/%Y"
            ).strftime("%Y-%m-%d")
            row["Amount"] = clean_amount(row["Amount"])
        except (ValueError, TypeError) as error:
            raise SystemExit(f"Line {line_number}: {error}")

        row["Transaction Date"] = row.pop("Date")
        writer.writerow(row)

print(f"Wrote {target}")
PY
```

target = "transactions_export_final.csv"

def clean_amount(value):
    text = (value or "").strip()
    if text.startswith("(") and text.endswith(")"):
        text = "-" + text[1:-1]

[assistant]
Use a dry-run mode first. This script checks `.jpg` and `.jpeg` files case-insensitively, keeps the original suffix exactly as written, sorts by capture date, and prints proposed changes without renaming anything.

Save it as `rename_photos.py` somewhere convenient:

```python
from pathlib import Path
from collections import defaultdict
from datetime import datetime
from PIL import Image
from PIL.ExifTags import TAGS

PHOTO_DIR = Path.home() / "Desktop" / "Beach Trip Photos"
DRY_RUN = True

DATE_TAGS = {"DateTimeOriginal", "DateTimeDigitized", "DateTime"}

def capture_date(path):
    try:
        with Image.open(path) as image:
            exif = image.getexif()
            values = {}
            for tag_id, value in exif.items():
                name = TAGS.get(tag_id, tag_id)
                values[name] = value

            for tag in DATE_TAGS:
                value = values.get(tag)
                if isinstance(value, str):
                    try:
                        return datetime.strptime(
                            value.strip(), "%Y:%m:%d %H:%M:%S"
                        )
                    except ValueError:
                        pass
    except Exception as error:
        print(f"SKIP {path.name}: {error}")

    return None

photos = []
for path in PHOTO_DIR.iterdir():
    if path.is_file() and path.suffix.lower() in {".jpg", ".jpeg"}:
        date = capture_date(path)
        if date is not None:
            photos.append((date, path))

photos.sort(key=lambda item: (item[0], item[1].name.lower()))

counters = defaultdict(int)
proposals = []

for date, path in photos:
    day = date.strftime("%Y-%m-%d")
    counters[day] += 1
    new_name = f"{day}_{counters[day]:03d}{path.suffix}"
    destination = path.with_name(new_name)
    proposals.append((path, destination))

for source, destination in proposals:
    if source.name == destination.name:
        print(f"KEEP {source.name}")
    elif destination.exists():
        print(f"CONFLICT {source.name} -> {destination.name} (already exists)")
    else:
        print(f"RENAME {source.name} -> {destination.name}")

if not DRY_RUN:
    for source, destination in proposals:
        if source.name != destination.name and not destination.exists():
            source.rename(destination)
```

Run the preview with:

```bash
python3 rename_photos.py
```


```bash
folder = Path.home() / "Desktop" / "Beach Trip Photos"

for path in sorted(folder.iterdir()):
    if path.is_file() and path.suffix.lower() in {".jpg", ".jpeg"}:
```

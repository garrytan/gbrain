Here is a command-line version that accepts `C`, `F`, `km`, or `mi`:

```python
#!/usr/bin/env python3
import sys

if len(sys.argv) != 3:
    print(f"Usage: {sys.argv[0]} C|F|km|mi value")
    raise SystemExit(2)

unit = sys.argv[1].lower()
try:
    value = float(sys.argv[2])
except ValueError:
    print("The value must be numeric.")
    raise SystemExit(2)

if unit == "c":
    result = value * 9 / 5 + 32
    print(f"{value:.2f} °C = {result:.2f} °F")
elif unit == "f":
    result = (value - 32) * 5 / 9
    print(f"{value:.2f} °F = {result:.2f} °C")
elif unit == "km":
    result = value * 0.6213711922
    print(f"{value:.2f} km = {result:.2f} miles")
elif unit == "mi":
    result = value * 1.609344
    print(f"{value:.2f} miles = {result:.2f} km")
else:
    print("Unit must be C, F, km, or mi.")
    raise SystemExit(2)
```

Use the blank cell as zero, then calculate `Credit - Debit`. This script reads the source file and writes a new CSV:

```python
import csv
from decimal import Decimal
from datetime import datetime

source_file = "bank_export.csv"
output_file = "budget_import.csv"

with open(source_file, newline="", encoding="utf-8-sig") as src, \
     open(output_file, "w", newline="", encoding="utf-8") as dst:

    reader = csv.DictReader(src)
    writer = csv.DictWriter(
        dst,
        fieldnames=["Date", "Description", "Amount", "Category"]
    )
    writer.writeheader()

    for row in reader:
        debit = Decimal((row["Debit"] or "0").replace(",", "").strip())
        credit = Decimal((row["Credit"] or "0").replace(",", "").strip())
        date = datetime.strptime(
            row["Transaction Date"].strip(), "%m/%d/%Y"
        ).date().isoformat()

        writer.writerow({
            "Date": date,
            "Description": row["Payee"].strip(),
            "Amount": f"{credit - debit:.2f}",
            "Category": row["Notes"].strip(),
        })
```

- Debit `42.18` becomes `-42.18`.
- Credit `4.75` becomes `4.75`.

debit = abs(debit_value)

Only use the `abs()` version if Debit is consistently an expense column and may occasionally carry a minus sign. Otherwise, retain the first version so the source signs are not changed unexpectedly.

Could you give me a complete script with validation? I’d like it to report rows with missing dates, invalid amounts, or both Debit and Credit filled in.

[assistant]
Here is a complete version. It skips invalid rows, writes them to `conversion_errors.csv`, and reports totals at the end.

```python
import csv
from decimal import Decimal, InvalidOperation
from datetime import datetime

SOURCE = "bank_export.csv"
OUTPUT = "budget_import.csv"
ERRORS = "conversion_errors.csv"

def parse_amount(value):
    text = (value or "").strip().replace("$", "").replace(",", "")
    if text == "":
        return Decimal("0.00")
    return Decimal(text)

def parse_date(value):
    text = (value or "").strip()
    if not text:
        raise ValueError("missing date")
    return datetime.strptime(text, "%m/%d/%Y").date().isoformat()

converted = 0
errors = []

with open(SOURCE, newline="", encoding="utf-8-sig") as src:
    reader = csv.DictReader(src)

    with open(OUTPUT, "w", newline="", encoding="utf-8") as dst:
        writer = csv.DictWriter(
            dst,
            fieldnames=["Date", "Description", "Amount", "Category"]
        )
        writer.writeheader()

        for line_number, row in enumerate(reader, start=2):
            problems = []

            try:
                date = parse_date(row.get("Transaction Date"))
            except (ValueError, TypeError):
                date = ""
                problems.append("invalid or missing date")

            try:
                debit_text = (row.get("Debit") or "").strip()
                credit_text = (row.get("Credit") or "").strip()

                if debit_text and credit_text:
                    problems.append("both Debit and Credit are filled")

                debit = parse_amount(debit_text)
                credit = parse_amount(credit_text)
                amount = credit - debit
            except (InvalidOperation, ValueError):
                amount = Decimal("0.00")
                problems.append("invalid amount")

            if problems:
                errors.append({
                    "Line": line_number,
                    "Problems": "; ".join(problems),
                    "SourceRow": repr(row),
                })
                continue

            writer.writerow({
                "Date": date,
                "Description": (row.get("Payee") or "").strip(),
                "Amount": f"{amount:.2f}",
                "Category": (row.get("Notes") or "").strip(),
            })
            converted += 1

with open(ERRORS, "w", newline="", encoding="utf-8") as err:
    writer = csv.DictWriter(err, fieldnames=["Line", "Problems", "SourceRow"])
    writer.writeheader()
    writer.writerows(errors)

print(f"Converted rows: {converted}")
print(f"Rows with errors: {len(errors)}")
print(f"Output file: {OUTPUT}")
print(f"Error file: {ERRORS}")
```

Run it with:

```text
python convert_csv.py
```

Example output:

```text
Converted rows: 294
Rows with errors: 6
Output file: budget_import.csv
Error file: conversion_errors.csv
```

Review `conversion_errors.csv`, correct those source rows, and rerun the script. Because the script overwrites the output files, each run starts cleanly.

Use a CSV-aware parser rather than `awk`, because a description containing a comma needs quoting. Python’s standard library can read the semicolon-delimited source and write valid comma-delimited CSV.

Create a file named `convert_budget.py`:

```python
import csv
from datetime import datetime

source = "march-export.csv"
destination = "march-budget.csv"

with open(source, "r", newline="", encoding="utf-8-sig") as infile, \
     open(destination, "w", newline="", encoding="utf-8") as outfile:

    reader = csv.DictReader(infile, delimiter=";")
    writer = csv.DictWriter(
        outfile,
        fieldnames=["Date", "Payee", "Amount", "Account"],
        lineterminator="\n"
    )

    writer.writeheader()

    for row in reader:
        original_date = row["Date"].strip()
        converted_date = datetime.strptime(
            original_date, "%m/%d/%Y"
        ).strftime("%Y-%m-%d")

        amount = row["Amount"].strip().replace(",", ".")

        writer.writerow({
            "Date": converted_date,
            "Payee": row["Description"].strip(),
            "Amount": amount,
            "Account": row["Account"].strip(),
        })
```

import csv
from datetime import datetime

source = "march-export.csv"
destination = "march-budget.csv"

with open(source, "r", newline="", encoding="utf-8-sig") as infile, \
     open(destination, "w", newline="", encoding="utf-8") as outfile:

    reader = csv.reader(infile, delimiter=";")
    writer = csv.DictWriter(
        outfile,
        fieldnames=["Date", "Payee", "Amount", "Account"],
        lineterminator="\n"
    )
    writer.writeheader()

    header = next(reader)

    for line_number, fields in enumerate(reader, start=2):
        if not any(field.strip() for field in fields):
            continue

        if len(fields) == 6 and fields[-1].strip() == "":
            fields = fields[:-1]

        if len(fields) != 5:
            print(f"Skipped line {line_number}: expected 5 fields, got {len(fields)}")
            continue

        date_text, payee, category, amount, account = fields

        try:
            date_text = datetime.strptime(
                date_text.strip(), "%m/%d/%Y"
            ).strftime("%Y-%m-%d")
        except ValueError:
            print(f"Skipped line {line_number}: invalid date {date_text!r}")
            continue

        writer.writerow({
            "Date": date_text,
            "Payee": payee.strip(),
            "Amount": amount.strip().replace(",", "."),
            "Account": account.strip(),
        })

Here is the complete script:

```python
import csv
from datetime import datetime

SOURCE = "march-export.csv"
DESTINATION = "march-budget.csv"

with open(SOURCE, "r", newline="", encoding="utf-8-sig") as infile, \
     open(DESTINATION, "w", newline="", encoding="utf-8") as outfile:

    reader = csv.reader(infile, delimiter=";")
    writer = csv.DictWriter(
        outfile,
        fieldnames=["Date", "Payee", "Amount", "Account"],
        lineterminator="\n"
    )

    header = next(reader, None)
    if header is None:
        raise SystemExit("The source file is empty.")

    writer.writeheader()

    for line_number, fields in enumerate(reader, start=2):
        if not any(field.strip() for field in fields):
            continue

        # Remove an empty extra field sometimes added at the end of a row.
        if len(fields) == 6 and fields[-1].strip() == "":
            fields = fields[:-1]

        if len(fields) != 5:
            print(
                f"Skipped line {line_number}: "
                f"expected 5 fields, got {len(fields)}"
            )
            continue

        date_text, payee, category, amount_text, account = fields

        try:
            output_date = datetime.strptime(
                date_text.strip(), "%m/%d/%Y"
            ).strftime("%Y-%m-%d")
        except ValueError:
            print(f"Skipped line {line_number}: invalid date {date_text!r}")
            continue

        try:
            amount = float(amount_text.strip().replace(",", "."))
        except ValueError:
            print(
                f"Skipped line {line_number}: "
                f"invalid amount {amount_text!r}"
            )
            continue

        if category.strip().lower() == "income":
            amount = abs(amount)
        else:
            amount = -abs(amount)

        writer.writerow({
            "Date": output_date,
            "Payee": payee.strip(),
            "Amount": f"{amount:.2f}",
            "Account": account.strip(),
        })

print(f"Wrote {DESTINATION}")
```

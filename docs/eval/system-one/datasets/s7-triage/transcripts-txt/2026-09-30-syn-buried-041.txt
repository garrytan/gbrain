Here is a Python script using only the standard library. It converts the dates and signed amounts, preserves commas safely, checks required fields, and writes a UTF-8 CSV.

```python
import csv
from datetime import datetime
from decimal import Decimal, InvalidOperation

INPUT_FILE = "bank_export.csv"
OUTPUT_FILE = "budgetwise_import.csv"
ACCOUNT_NAME = "Checking"

required_columns = {
    "Posted Date", "Description", "Debit", "Credit", "Category"
}

def parse_amount(value, row_number, field_name):
    value = value.strip().replace(",", "")
    if not value:
        return None
    try:
        amount = Decimal(value)
    except InvalidOperation:
        raise ValueError(
            f"Row {row_number}: {field_name} is not numeric: {value!r}"
        )
    if amount < 0:
        raise ValueError(
            f"Row {row_number}: {field_name} cannot be negative: {value!r}"
        )
    return amount

def format_amount(amount):
    return f"{amount:.2f}"

errors = []
converted = []

with open(INPUT_FILE, "r", encoding="utf-8-sig", newline="") as source:
    reader = csv.DictReader(source)

    if reader.fieldnames is None:
        raise SystemExit("The input file has no header row.")

    missing = required_columns - set(reader.fieldnames)
    if missing:
        raise SystemExit(
            "Missing required columns: " + ", ".join(sorted(missing))
        )

    for row_number, row in enumerate(reader, start=2):
        try:
            raw_date = row["Posted Date"].strip()
            try:
                date = datetime.strptime(raw_date, "%m/%d/%Y").strftime("%Y-%m-%d")
            except ValueError:
                raise ValueError(
                    f"Row {row_number}: invalid date {raw_date!r}; "
                    "expected MM/DD/YYYY"
                )

            payee = row["Description"].strip()
            if not payee:
                raise ValueError(f"Row {row_number}: Description is blank")

            debit = parse_amount(row["Debit"], row_number, "Debit")
            credit = parse_amount(row["Credit"], row_number, "Credit")

            if debit is not None and credit is not None:
                raise ValueError(
                    f"Row {row_number}: both Debit and Credit have values"
                )
            if debit is None and credit is None:
                raise ValueError(
                    f"Row {row_number}: both Debit and Credit are blank"
                )

            amount = -debit if debit is not None else credit
            category = row["Category"].strip()

            converted.append({
                "Date": date,
                "Payee": payee,
                "Amount": format_amount(amount),
                "Category": category,
                "Notes": "",
                "Account": ACCOUNT_NAME,
            })

        except (KeyError, ValueError) as exc:
            errors.append(str(exc))

if errors:
    print(f"Validation failed with {len(errors)} error(s):")
    for error in errors:
        print(f"- {error}")
    raise SystemExit(1)

with open(OUTPUT_FILE, "w", encoding="utf-8", newline="") as target:
    fieldnames = ["Date", "Payee", "Amount", "Category", "Notes", "Account"]
    writer = csv.DictWriter(target, fieldnames=fieldnames)
    writer.writeheader()
    writer.writerows(converted)

print(f"Converted {len(converted)} rows to {OUTPUT_FILE}")
```

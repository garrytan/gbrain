Yes. Python’s `csv` module will correctly handle commas and escaped quotation marks, provided the bank export is a valid CSV. This script reads `bank-export.csv` and writes an app-ready file with `Date,Payee,Amount`.

Save it as `convert_transactions.py`:

```python
import csv
from datetime import datetime
from decimal import Decimal, InvalidOperation

INPUT_FILE = "bank-export.csv"
OUTPUT_FILE = "budget-import.csv"

def clean_money(value, row_number, field_name):
    value = (value or "").strip()
    if not value:
        return None

    # Remove a currency symbol and thousands separators if the bank includes them.
    value = value.replace("$", "").replace(",", "")

    try:
        return Decimal(value)
    except InvalidOperation:
        raise ValueError(
            f"Row {row_number}: invalid {field_name} value {value!r}"
        )

with open(INPUT_FILE, "r", encoding="utf-8-sig", newline="") as source, \
     open(OUTPUT_FILE, "w", encoding="utf-8", newline="") as target:

    reader = csv.DictReader(source)

    required = {"Date", "Description", "Debit", "Credit"}
    missing = required - set(reader.fieldnames or [])
    if missing:
        raise ValueError(f"Missing required columns: {sorted(missing)}")

    writer = csv.DictWriter(
        target,
        fieldnames=["Date", "Payee", "Amount"],
        quoting=csv.QUOTE_MINIMAL
    )
    writer.writeheader()

    for row_number, row in enumerate(reader, start=2):
        raw_date = (row.get("Date") or "").strip()
        description = (row.get("Description") or "").strip()
        debit = clean_money(row.get("Debit"), row_number, "Debit")
        credit = clean_money(row.get("Credit"), row_number, "Credit")

        if not raw_date:
            raise ValueError(f"Row {row_number}: missing Date")
        if not description:
            raise ValueError(f"Row {row_number}: missing Description")
        if debit is not None and credit is not None:
            raise ValueError(
                f"Row {row_number}: both Debit and Credit contain values"
            )
        if debit is None and credit is None:
            raise ValueError(
                f"Row {row_number}: both Debit and Credit are blank"
            )

        try:
            converted_date = datetime.strptime(
                raw_date, "%d/%m/%Y"
            ).strftime("%Y-%m-%d")
        except ValueError:
            raise ValueError(
                f"Row {row_number}: expected DD/MM/YYYY, got {raw_date!r}"
            )

        amount = -debit if debit is not None else credit

        writer.writerow({
            "Date": converted_date,
            "Payee": description,
            "Amount": f"{amount:.2f}",
        })

print(f"Wrote {OUTPUT_FILE}")
```

This deliberately stops on ambiguous or incomplete rows instead of silently producing questionable transactions.

For a full-file workflow that continues past bad rows, use this revised script. It writes valid rows to `budget-import.csv` and review rows, including the reason, to `budget-review.csv`.

```python
import csv
from datetime import datetime
from decimal import Decimal, InvalidOperation

INPUT_FILE = "bank-export.csv"
OUTPUT_FILE = "budget-import.csv"
REVIEW_FILE = "budget-review.csv"

def money(value):
    value = (value or "").strip().replace("$", "").replace(",", "")
    if not value:
        return None
    try:
        return Decimal(value)
    except InvalidOperation:
        return "INVALID"

with open(INPUT_FILE, "r", encoding="utf-8-sig", newline="") as source, \
     open(OUTPUT_FILE, "w", encoding="utf-8", newline="") as good_file, \
     open(REVIEW_FILE, "w", encoding="utf-8", newline="") as review_file:

    reader = csv.DictReader(source)
    required = {"Date", "Description", "Debit", "Credit"}
    missing = required - set(reader.fieldnames or [])
    if missing:
        raise ValueError(f"Missing required columns: {sorted(missing)}")

    good_writer = csv.DictWriter(
        good_file,
        fieldnames=["Date", "Payee", "Amount"],
        quoting=csv.QUOTE_MINIMAL
    )
    good_writer.writeheader()

    review_fields = ["SourceRow", "Reason"] + reader.fieldnames
    review_writer = csv.DictWriter(
        review_file,
        fieldnames=review_fields,
        quoting=csv.QUOTE_MINIMAL
    )
    review_writer.writeheader()

    good_count = 0
    review_count = 0

    for row_number, row in enumerate(reader, start=2):
        reason = None
        raw_date = (row.get("Date") or "").strip()
        description = (row.get("Description") or "").strip()
        debit = money(row.get("Debit"))
        credit = money(row.get("Credit"))

        if not raw_date:
            reason = "missing Date"
        elif not description:
            reason = "missing Description"
        elif debit == "INVALID":
            reason = "invalid Debit"
        elif credit == "INVALID":
            reason = "invalid Credit"
        elif debit is not None and credit is not None:
            reason = "both Debit and Credit are populated"
        elif debit is None and credit is None:
            reason = "both Debit and Credit are blank"

        converted_date = None
        if reason is None:
            try:
                converted_date = datetime.strptime(
                    raw_date, "%d/%m/%Y"
                ).strftime("%Y-%m-%d")
            except ValueError:
                reason = "invalid Date; expected DD/MM/YYYY"

        if reason is not None:
            review_writer.writerow({
                "SourceRow": row_number,
                "Reason": reason,
                **row
            })
            review_count += 1
            continue

        amount = -debit if debit is not None else credit
        good_writer.writerow({
            "Date": converted_date,
            "Payee": description,
            "Amount": f"{amount:.2f}",
        })
        good_count += 1

print(f"Wrote {good_count} importable rows to {OUTPUT_FILE}")
print(f"Wrote {review_count} rows for review to {REVIEW_FILE}")
```

You can create a report of exact duplicates in the converted file with:

```python
import csv

groups = defaultdict(list)

with open("budget-import.csv", encoding="utf-8", newline="") as f:
    for line_number, row in enumerate(csv.DictReader(f), start=2):
        key = (row["Date"], row["Payee"].strip().casefold(), row["Amount"])

for key, lines in groups.items():
```

```bash
#!/bin/bash
# node_modules_cleaner.sh
# Finds and optionally removes node_modules from inactive projects

PROJECTS_DIR="$HOME/Projects"
DAYS_INACTIVE=90

echo "Scanning for node_modules directories..."
echo "========================================="

find "$PROJECTS_DIR" -name "node_modules" -type d -prune | while read -r nm_path; do
    project_dir=$(dirname "$nm_path")
    size=$(du -sh "$nm_path" 2>/dev/null | cut -f1)
    
    # Check last modification time of package.json
    pkg_json="$project_dir/package.json"
    if [ -f "$pkg_json" ]; then
        last_mod=$(stat -f "%m" "$pkg_json" 2>/dev/null || stat -c "%Y" "$pkg_json" 2>/dev/null)
        current=$(date +%s)
        days_old=$(( (current - last_mod) / 86400 ))
        
        if [ $days_old -gt $DAYS_INACTIVE ]; then
            echo "[STALE] $nm_path"
            echo "        Size: $size | Last modified: $days_old days ago"
        else
            echo "[ACTIVE] $nm_path"
            echo "        Size: $size | Last modified: $days_old days ago"
        fi
    fi
done

echo ""
echo "To remove stale node_modules, run:"
echo "find \$PROJECTS_DIR -name 'node_modules' -type d -prune -exec rm -rf {} +"
```

```bash
#!/bin/bash
# quarterly_cleanup.sh
# Run this every few months to keep disk bloat in check

echo "=== Quarterly Disk Cleanup Script ==="
echo ""

# Show current disk usage
echo "Current disk status:"
df -h / | tail -1
echo ""

# Clean package manager caches
echo "Cleaning package manager caches..."
brew cleanup --prune=all 2>/dev/null && echo "  Homebrew: done"
pip cache purge 2>/dev/null && echo "  pip: done"
yarn cache clean 2>/dev/null && echo "  yarn: done"
npm cache clean --force 2>/dev/null && echo "  npm: done"

# Find large node_modules in projects older than 90 days
echo ""
echo "Stale node_modules (not modified in 90+ days):"
find ~/Projects -name "node_modules" -type d -prune -mtime +90 -exec du -sh {} \; 2>/dev/null

# Find old installers in Downloads
echo ""
echo "Old installers in Downloads (180+ days):"
find ~/Downloads -type f \( -name "*.dmg" -o -name "*.pkg" -o -name "*.xip" \) -mtime +180 -exec ls -lh {} \; 2>/dev/null

# Show trash size
echo ""
echo "Trash size:"
du -sh ~/.Trash 2>/dev/null || echo "  Empty or inaccessible"

# Clean unavailable simulators if Xcode is installed
if command -v xcrun &> /dev/null; then
    echo ""
    echo "Cleaning unavailable simulators..."
    xcrun simctl delete unavailable 2>/dev/null
fi

echo ""
echo "=== Cleanup suggestions complete ==="
```
